#include "ft_traceroute.h"

int main(int argc, char **argv)
{
    t_traceroute data;

    memset(&data, 0, sizeof(t_traceroute));
    
    parse_args(&data, argc, argv);

    init_sockets(&data);

    printf("traceroute to %s (%s), %d hops max, 60 bytes packets\n", 
           data.hostname, data.ip_str, MAX_TTL);

    run_traceroute(&data);

    close(data.udp_sockfd);
    close(data.icmp_sockfd);
    return (0);
}


net.c

#include "ft_traceroute.h"

void print_usage(void)
{
    printf("Usage: ./ft_traceroute [OPTION...] HOST...\n");
    printf("Print the route packet trace to network host.\n\n");
    printf("--help              Give this help list\n");
    printf("-m Max_ttl          Set the max number of hops (max TTL to be reached). Default is 30\n");
    printf("-q nqueries         Set the number of probes per each hop. Default is 3\n");
    printf("-p port             Set the destination port to use. It is either initial udp port value\n for \"default\" method (incremented by each probe, default is 33434),\n or initial seq for \"icmp\" (incremented as well, default from 1),\nor some constant destination port for other methods (with default of 80 \nfor \"tcp\", 53 for \"udp\", etc.).\n");
    printf("-w wait             Wait for a probe no more than wait seconds (default 1)\n");
    printf("-f first_ttl        Start from the first_ttl hop (instead from 1)\n");
    exit(EXIT_SUCCESS);
}

void resolve_hostname(t_traceroute *data, const char *host)
{
    struct addrinfo hints, *result;

    memset(&hints, 0, sizeof(hints));
    hints.ai_family = AF_INET;
    hints.ai_socktype = SOCK_DGRAM;

    int ret = getaddrinfo(host, NULL, &hints, &result);
    if (ret != 0)
    {
        fprintf(stderr, "ft_traceroute: unknown host %s\n", host);
        exit(EXIT_FAILURE);
    }

    struct sockaddr_in *sa = (struct sockaddr_in *)result->ai_addr;
    memcpy(&data->target_sa, sa, sizeof(struct sockaddr_in));
    
    char *ip_temp = inet_ntoa(sa->sin_addr);
    if (ip_temp)
    {
        strncpy(data->ip_str, ip_temp, INET_ADDRSTRLEN);
        data->ip_str[INET_ADDRSTRLEN - 1] = '\0';
    }

    freeaddrinfo(result);
}

void parse_args(t_traceroute *data, int argc, char **argv)
{
    if (getuid() != 0)
    {
        fprintf(stderr, "ft_traceroute: root privileges are required.\n");
        exit(EXIT_FAILURE);
    }
    if (argc < 2)
    {
        print_usage();
    }

    data->max_ttl = 30;
    data->nqueries = 3;
    data->base_port = 33434;
    data->wait_time = 1;
    data->first_ttl = 1;
    data->hostname = NULL;

    for (int i = 1; i < argc; i++)
    {
        if (strcmp(argv[i], "--help") == 0)
        {
            print_usage();
        }
        else if (strcmp(argv[i], "-m") == 0 && i + 1 < argc)
        {
            data->max_ttl = atoi(argv[++i]);
        }
        else if (strcmp(argv[i], "-q") == 0 && i + 1 < argc)
        {
            data->nqueries = atoi(argv[++i]);
        }
        else if (strcmp(argv[i], "-p") == 0 && i + 1 < argc)
        {
            data->base_port = atoi(argv[++i]);
        }
        else if (strcmp(argv[i], "-w") == 0 && i + 1 < argc)
        {
            data->wait_time = atoi(argv[++i]);
        }
        else if (strcmp(argv[i], "-f") == 0 && i + 1 < argc)
        {
            data->first_ttl = atoi(argv[++i]);
        }
        else if (argv[i][0] == '-')
        {
            fprintf(stderr, "ft_traceroute: invalid option -- '%s'\n", argv[i] + 1);
            print_usage();
        }
        else
            if (data->hostname != NULL)

            {
                fprintf(stderr, "ft_traceroute: too many arguments\n");
                print_usage();
            }
                data->hostname = argv[i];
    }

    if (!data->hostname)
    {
        fprintf(stderr, "ft_traceroute: missing host operand\n");
        print_usage();
    }

    resolve_hostname(data, data->hostname);
}

void init_sockets(t_traceroute *data)
{
    data->udp_sockfd = socket(AF_INET, SOCK_DGRAM, IPPROTO_UDP);
    if (data->udp_sockfd < 0)
    {
        perror("ft_traceroute: UDP socket creation failed");
        exit(EXIT_FAILURE);
    }

    data->icmp_sockfd = socket(AF_INET, SOCK_RAW, IPPROTO_ICMP);
    if (data->icmp_sockfd < 0)
    {
        perror("ft_traceroute: ICMP socket creation failed");
        close(data->udp_sockfd);
        exit(EXIT_FAILURE);
    }
}


traceroute.c

#include "ft_traceroute.h"

static double calcul_rtt(struct timeval *start, struct timeval *end)
{
    double s = (double)(end->tv_sec - start->tv_sec) * 1000.0;
    double us = (double)(end->tv_usec - start->tv_usec) / 1000.0;
    return (s + us);
}

static int parse_icmp_response(char *buffer, ssize_t bytes, uint16_t expected_port, int *target_reached)
{
    struct iphdr *ip = (struct iphdr *)buffer;
    int ip_len = ip->ihl * 4;

    if (bytes < ip_len + (ssize_t)sizeof(struct icmphdr))
        return (0);

    struct icmphdr *icmp = (struct icmphdr *)(buffer + ip_len);
    struct iphdr *orig_ip = (struct iphdr *)(buffer + ip_len + sizeof(struct icmphdr));
    int orig_ip_len = orig_ip->ihl * 4;
    
    struct udphdr *orig_udp = (struct udphdr *)(buffer + ip_len + sizeof(struct icmphdr) + orig_ip_len);

    if (ntohs(orig_udp->dest) != expected_port)
        return (0);

    if (icmp->type == ICMP_DEST_UNREACH)
    {
        *target_reached = 1;
        return (1);
    }
    else if (icmp->type == ICMP_TIME_EXCEEDED)
    {
        return (1);
    }

    return (0);
}

void send_probe(t_traceroute *data, struct timeval *send_time)
{
    char payload[32];

    memset(payload, 42, sizeof(payload));
    data->target_sa.sin_port = htons(data->udp_port);

    if (gettimeofday(send_time, NULL) < 0)
    {
        perror("ft_traceroute: gettimeofday failed");
        exit(EXIT_FAILURE);
    }

    if (sendto(data->udp_sockfd, payload, sizeof(payload), 0, 
               (struct sockaddr *)&data->target_sa, sizeof(data->target_sa)) < 0)
    {
        perror("ft_traceroute: sendto failed");
    }
    data->udp_port++;
}

int receive_probe(t_traceroute *data, struct timeval *send_time, char *last_ip, uint16_t expected_port, int *target_reached)
{
    fd_set          rfds;
    struct timeval  tv;
    struct timeval  recv_time;
    struct sockaddr_in recv_sa;
    socklen_t       addr_len = sizeof(recv_sa);
    char            buffer[512];
    char            current_ip[INET_ADDRSTRLEN];
    
    while (1)
    {
        FD_ZERO(&rfds);
        FD_SET(data->icmp_sockfd, &rfds);

        tv.tv_sec = data->wait_time;
        tv.tv_usec = 0;

        int ret = select(data->icmp_sockfd + 1, &rfds, NULL, NULL, &tv);
        if (ret < 0)
        {
            perror("ft_traceroute: select failed");
            return (-1);
        }
        else if (ret == 0)
        {
            printf(" *");
            return (0);
        }

        ssize_t bytes_received = recvfrom(data->icmp_sockfd, buffer, sizeof(buffer), 
                                          0, (struct sockaddr *)&recv_sa, &addr_len);
        if (bytes_received < 0)
        {
            perror("ft_traceroute: recvfrom failed");
            return (-1);
        }

        gettimeofday(&recv_time, NULL);

        if (parse_icmp_response(buffer, bytes_received, expected_port, target_reached))
        {
            char *ip_temp = inet_ntoa(recv_sa.sin_addr);
            if (ip_temp)
            {
                strncpy(current_ip, ip_temp, sizeof(current_ip));
                current_ip[sizeof(current_ip) - 1] = '\0';
            }
            else
            {
                strcpy(current_ip, "unknown");
            }

            if (strcmp(last_ip, current_ip) != 0)
            {
                printf(" %s", current_ip);
                strncpy(last_ip, current_ip, INET_ADDRSTRLEN);
                last_ip[INET_ADDRSTRLEN - 1] = '\0';
            }
            double rtt = calcul_rtt(send_time, &recv_time);
            printf("  %.3f ms", rtt);
            return (1);
        }
    }
}

void run_traceroute(t_traceroute *data)
{
    struct timeval send_times[data->nqueries > 0 ? data->nqueries : 1];
    int target_reached = 0;

    data->udp_port = data->base_port;

    for (data->curttl = data->first_ttl; data->curttl <= data->max_ttl && !target_reached; data->curttl++)
    {
        if (setsockopt(data->udp_sockfd, IPPROTO_IP, IP_TTL, &data->curttl, sizeof(data->curttl)) < 0)
        {
            perror ("ft_traceroute: setsockopt IP_TTL failed");
            exit(EXIT_FAILURE);
        }
        printf("%2d", data->curttl);
        fflush(stdout);

        char last_ip[INET_ADDRSTRLEN] = "";

        for(int i = 0; i < data->nqueries && !target_reached; i++)
        {
            uint16_t current_probe_port = data->udp_port;

            send_probe(data, &send_times[i]);
            receive_probe(data, &send_times[i], last_ip, current_probe_port, &target_reached);
            fflush(stdout);
        }
        printf("\n");
    } 
}
